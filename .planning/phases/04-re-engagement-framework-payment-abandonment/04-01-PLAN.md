    Logic per Decision 4 in 04-CONTEXT.md:
    1. Trim whitespace; collapse internal newlines (`\n`, `\r`) to single spaces; collapse runs of spaces to one space.
    2. Try to split on the regex `/[.!?]\s/` and take the first segment (the first sentence).
    3. If the first-sentence segment is `<= 120` chars → return it as-is (no trailing period — the original split removed it).
    4. If the first-sentence segment is `> 120` chars → return `slice(0, 117) + "..."`.
    5. If no `[.!?]\s` match exists at all (a single long sentence with no punctuation) AND the input is `> 120` chars → return `slice(0, 117) + "..."`.
    6. If no `[.!?]\s` match exists AND the input is `<= 120` chars → return the trimmed input verbatim.

    Edge cases tests must cover:

    - Whitespace-only string → `""`
    - Single sentence ending with `.` followed by space and second sentence → returns first segment WITHOUT the trailing punctuation

    - Multiline description with `\n` between sentences → newlines normalised to spaces before splitting
    - Exactly 120 chars, no punctuation → returned verbatim (NO ellipsis)
    - 121 chars, no punctuation → returned as `slice(0, 117) + "..."` (length exactly 120)
    - First sentence is 200 chars → returned as `slice(0, 117) + "..."`

    ```typescript
    /**
     * Format a matter description for inline rendering in re-engagement emails.
     * Implements 04-CONTEXT.md Decision 4: first sentence (split on [.!?] + whitespace),
     * or first 120 chars + ellipsis if no sentence-ending punctuation. Newlines
     * collapsed to spaces; whitespace runs normalised; never throws.
     *
     * Used by: src/lib/email/templates/reengagement-payment.tsx (1h and 24h variants).
     */
    const MAX_SNIPPET_LENGTH = 120;
    const TRUNCATE_AT = 117; // 120 - "...".length

    export function snippetMatter(matterDescription: string): string {
      if (!matterDescription) return "";

      // Normalise whitespace: collapse \n, \r, runs of spaces.
      const normalised = matterDescription
        .replace(/[\r\n]+/g, " ")
        .replace(/\s+/g, " ")
        .trim();

      if (normalised.length === 0) return "";

      // Split on sentence-ending punctuation followed by whitespace.
      // The split keeps the first segment without the trailing [.!?].
      const parts = normalised.split(/[.!?]\s/);
      const firstSentence = parts[0] ?? normalised;

      if (firstSentence.length <= MAX_SNIPPET_LENGTH) return firstSentence;
      return firstSentence.slice(0, TRUNCATE_AT) + "...";
    }
    ```
